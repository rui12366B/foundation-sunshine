cmake_minimum_required(VERSION 3.20)
project(foundation_console_helper_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
enable_testing()
option(ENABLE_SANITIZERS "ASan and UBSan for portable tests" OFF)
foreach(name protocol timer)
  add_executable(${name}_test ${name}_test.cpp)
  target_include_directories(${name}_test PRIVATE "${ROOT}")
  target_link_libraries(${name}_test PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(${name}_test PRIVATE /W4 /EHsc)
  else()
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ENABLE_SANITIZERS)
      target_compile_options(${name}_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${name}_test PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
if(WIN32)
  add_library(display_bridge_client STATIC "${ROOT}/src/platform/windows/display_session_bridge/client.cpp")
  target_include_directories(display_bridge_client PUBLIC "${ROOT}")
  target_compile_definitions(display_bridge_client PUBLIC UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(display_bridge_client PUBLIC user32 advapi32 wtsapi32 userenv bcrypt)
  add_executable(foundation_display_helper "${ROOT}/tools/display_session_helper.cpp" "${ROOT}/src/platform/windows/display_session_bridge/dispatch.cpp")
  target_include_directories(foundation_display_helper PRIVATE "${ROOT}")
  target_compile_definitions(foundation_display_helper PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(foundation_display_helper PRIVATE user32 advapi32 wtsapi32)
  if(MINGW)
    target_link_options(foundation_display_helper PRIVATE -municode -static)
  endif()
  add_executable(bridge_probe probe_windows.cpp)
  target_link_libraries(bridge_probe PRIVATE display_bridge_client)
  add_dependencies(bridge_probe foundation_display_helper)
  add_custom_command(TARGET bridge_probe POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:bridge_probe>/tools"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "$<TARGET_FILE:foundation_display_helper>" "$<TARGET_FILE_DIR:bridge_probe>/tools/")
  add_executable(transport_test transport_windows_test.cpp "${ROOT}/src/platform/windows/display_session_bridge/dispatch.cpp")
  target_include_directories(transport_test PRIVATE "${ROOT}")
  target_compile_definitions(transport_test PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(transport_test PRIVATE user32 advapi32 wtsapi32 Threads::Threads)
  add_test(NAME win32_transport COMMAND transport_test)
  set_tests_properties(win32_transport PROPERTIES TIMEOUT 30)
endif()
